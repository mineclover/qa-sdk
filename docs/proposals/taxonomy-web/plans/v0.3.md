```typescript
import { useMemo } from "react";
import { parseRef } from "taxonomy-engine";
import type { StaticTaxonomy } from "../types/taxonomy";

export function CoordinateFlowView({
  selectedRef,
  taxonomy,
  onSelectRef,
}: {
  selectedRef: string;
  taxonomy: StaticTaxonomy;
  onSelectRef: (ref: string) => void;
}) {
  const flows = useMemo(() => {
    let ref;
    try { ref = parseRef(selectedRef); } catch { return []; }
    if (ref.kind !== "entry") return [];

    // 선택 entry가 등장하는 모든 relation rule에서 출발 → 도달 가능 좌표
    const result: Array<{ relation: string; via: string; reachable: Array<Record<string, string>> }> = [];
    for (const [relName, rel] of Object.entries(taxonomy.relations)) {
      // 시작 좌표가 이 relation 안에 있는지
      const startCoord: Record<string, string> = { [ref.axis]: ref.entry };
      // 부분좌표 매칭: rule이 startCoord의 모든 key-value를 포함하면 호환
      const visited = new Set<string>();
      const stack: Array<Record<string, string>> = [startCoord];
      const reachable: Array<Record<string, string>> = [];

      while (stack.length > 0) {
        const cur = stack.pop()!;
        const key = Object.entries(cur).sort().map(([k, v]) => `${k}=${v}`).join("|");
        if (visited.has(key)) continue;
        visited.add(key);
        reachable.push(cur);

        for (const rule of rel.rules) {
          // cur의 모든 key-value가 rule에 일치하는가
          const compatible = Object.entries(cur).every(([k, v]) => rule[k] === v);
          if (!compatible) continue;
          const next: Record<string, string> = { ...rule };
          const nextKey = Object.entries(next).sort().map(([k, v]) => `${k}=${v}`).join("|");
          if (!visited.has(nextKey)) stack.push(next);
        }
      }

      if (reachable.length > 1) {
        result.push({
          relation: relName,
          via: rel.kind,
          reachable: reachable.slice(1),  // 시작점 제외
        });
      }
    }
    return result;
  }, [selectedRef, taxonomy]);

  if (flows.length === 0) {
    return <div className="text-xs text-gray-500">cross-axis 흐름 없음 (선택 entry가 등장하는 relation에서 다른 좌표로 연결되지 않음)</div>;
  }

  return (
    <ul className="space-y-2 text-xs">
      {flows.map((f) => (
        <li key={f.relation} className="border-l-2 border-purple-200 pl-2">
          <div className="font-mono text-purple-700">{f.relation} ({f.via})</div>
          <ul className="ml-2 mt-1 space-y-0.5">
            {f.reachable.map((coord, i) => {
              const display = Object.entries(coord).map(([k, v]) => `${k}=${v}`).join(", ");
              // entry 좌표면 클릭 가능
              const isSingle = Object.keys(coord).length === 1;
              const ref = isSingle
                ? `${taxonomy.taxonomy}:${Object.keys(coord)[0]}.${Object.values(coord)[0]}`
                : null;
              return (
                <li key={i}>
                  {ref ? (
                    <button className="font-mono text-blue-600 hover:underline" onClick={() => onSelectRef(ref)}>
                      {display}
                    </button>
                  ) : (
                    <span className="font-mono text-gray-600">{display}</span>
                  )}
                </li>
              );
            })}
          </ul>
        </li>
      ))}
    </ul>
  );
}
```

```typescript
import type { StaticTaxonomy, TaxonomyIndex } from "../types/taxonomy";
import type { Feedback } from "../types/feedback";

export type SearchHit =
  | { kind: "taxonomy"; name: string; file: string; description?: string }
  | { kind: "axis"; taxonomy: string; axis: string; entriesCount: number }
  | { kind: "entry"; taxonomy: string; axis: string; entry: string; label?: string }
  | { kind: "feedback"; id: string; ref: string; text: string };

export function buildSearchIndex(
  taxonomies: Array<{ idx: TaxonomyIndex["taxonomies"][number]; data: StaticTaxonomy }>,
  feedbacks: Feedback[],
): SearchHit[] {
  const hits: SearchHit[] = [];

  for (const { idx, data } of taxonomies) {
    hits.push({ kind: "taxonomy", name: data.taxonomy, file: idx.file, description: data.description });
    for (const [axisName, axis] of Object.entries(data.axes)) {
      hits.push({ kind: "axis", taxonomy: data.taxonomy, axis: axisName, entriesCount: axis.entries.length });
      for (const entry of axis.entries) {
        hits.push({ kind: "entry", taxonomy: data.taxonomy, axis: axisName, entry: entry.id, label: entry.label });
      }
    }
  }

  for (const f of feedbacks) {
    hits.push({ kind: "feedback", id: f.id, ref: f.ref, text: f.text });
  }

  return hits;
}

export function searchHits(hits: SearchHit[], query: string, max = 30): SearchHit[] {
  const q = query.toLowerCase().trim();
  if (!q) return [];
  return hits
    .filter((h) => {
      if (h.kind === "taxonomy") return h.name.toLowerCase().includes(q) || h.description?.toLowerCase().includes(q);
      if (h.kind === "axis") return h.axis.toLowerCase().includes(q);
      if (h.kind === "entry") return h.entry.toLowerCase().includes(q) || h.label?.toLowerCase().includes(q);
      if (h.kind === "feedback") return h.ref.toLowerCase().includes(q) || h.text.toLowerCase().includes(q);
      return false;
    })
    .slice(0, max);
}
```

```typescript
import { useState } from "react";

export function SearchBar({ onNavigate }: { onNavigate: (target: { file?: string; ref?: string }) => void }) {
  const [query, setQuery] = useState("");
  const [open, setOpen] = useState(false);

  return (

          setQuery(e.target.value);

        }}

      {open && (
        <SearchResults
          query={query}
          onClose={() => setOpen(false)}
          onNavigate={(t) => {
            onNavigate(t);
            setOpen(false);
            setQuery("");
          }}
        />
      )}
    </div>
  );
}
```

`SearchResults.tsx`:

```typescript
import { useEffect, useState } from "react";
import type { SearchHit } from "../lib/searchIndex";
import { buildSearchIndex, searchHits } from "../lib/searchIndex";
import { loadTaxonomyIndex, loadTaxonomy } from "../data/loadStaticTaxonomies";
import { fetchFeedbacks } from "../data/feedbacksApi";

let cachedHits: SearchHit[] | null = null;

async function getHits(): Promise<SearchHit[]> {
  if (cachedHits) return cachedHits;
  const index = await loadTaxonomyIndex();
  const taxonomies = await Promise.all(
    index.taxonomies.map(async (idx) => ({ idx, data: await loadTaxonomy(idx.file) })),
  );
  const feedbacks = await fetchFeedbacks();
  cachedHits = buildSearchIndex(taxonomies, feedbacks);
  return cachedHits;
}

export function SearchResults({
  query,
  onClose,
  onNavigate,
}: {
  query: string;
  onClose: () => void;
  onNavigate: (target: { file?: string; ref?: string }) => void;
}) {
  const [results, setResults] = useState<SearchHit[]>([]);

  useEffect(() => {
    getHits().then((hits) => setResults(searchHits(hits, query)));
  }, [query]);

  if (results.length === 0) {
    return (
      <div className="absolute right-0 top-full mt-1 w-96 max-h-96 overflow-auto bg-white border rounded shadow-lg p-3 z-50 text-sm">
        결과 없음
      </div>
    );
  }

  return (
    <div className="absolute right-0 top-full mt-1 w-96 max-h-96 overflow-auto bg-white border rounded shadow-lg z-50 text-sm">
      <ul className="divide-y">
        {results.map((h, i) => (
          <li
            key={i}
            className="p-2 cursor-pointer hover:bg-gray-50"
            onClick={() => {
              if (h.kind === "taxonomy") onNavigate({ file: h.file });
              if (h.kind === "axis") onNavigate({ ref: `${h.taxonomy}:${h.axis}` });
              if (h.kind === "entry") onNavigate({ ref: `${h.taxonomy}:${h.axis}.${h.entry}` });
              if (h.kind === "feedback") onNavigate({ ref: h.ref });
            }}
          >
            <div className="text-xs text-gray-500">{h.kind}</div>
            <div className="font-mono">
              {h.kind === "taxonomy" && h.name}
              {h.kind === "axis" && `${h.taxonomy}:${h.axis} (${h.entriesCount})`}
              {h.kind === "entry" && `${h.taxonomy}:${h.axis}.${h.entry}`}
              {h.kind === "feedback" && h.ref}
            </div>
            {h.kind === "entry" && h.label && <div className="text-xs text-gray-600">{h.label}</div>}
            {h.kind === "feedback" && <div className="text-xs text-gray-600 truncate">{h.text}</div>}
            {h.kind === "taxonomy" && h.description && <div className="text-xs text-gray-600 truncate">{h.description}</div>}
          </li>
        ))}
      </ul>
    </div>
  );
}
```
