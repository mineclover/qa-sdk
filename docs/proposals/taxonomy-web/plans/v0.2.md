```typescript
import type { Plugin } from "vite";
import { execSync } from "node:child_process";

type GitCommit = {
  sha: string;
  author: string;
  date: string;
  subject: string;
};

export function fetchGitLog(opts: {
  cwd: string;
  file: string;
  limit?: number;
  searchTerm?: string;
}): GitCommit[] {
  const limit = opts.limit ?? 50;
  const args = ["log", `--max-count=${limit}`, "--pretty=format:%H|%an|%ai|%s"];
  if (opts.searchTerm) {
    args.push(`-S${opts.searchTerm}`);
  }
  args.push("--", opts.file);
  try {
    const out = execSync(`git ${args.map((a) => `"${a.replace(/"/g, '\\"')}"`).join(" ")}`, {
      cwd: opts.cwd,
      encoding: "utf8",
      stdio: ["pipe", "pipe", "pipe"],
    }).trim();
    if (!out) return [];
    return out.split("\n").map((line) => {
      const [sha, author, date, subject] = line.split("|");
      return { sha, author, date, subject };
    });
  } catch {
    return [];
  }
}

export function gitLogApiPlugin(repoRoot: string): Plugin {
  return {
    name: "taxonomy-web:git-log-api",
    configureServer(server) {
      server.middlewares.use("/api/git-log", (req, res, next) => {
        if (req.method !== "GET") {
          next();
          return;
        }
        const url = new URL(req.url ?? "/", "http://localhost");
        const file = url.searchParams.get("file");
        const ref = url.searchParams.get("ref");
        if (!file) {
          res.statusCode = 400;
          res.end("file query required");
          return;
        }
        // ref에서 entry id 추출 (가장 단순한 형식 "tax:axis.entry")
        let searchTerm: string | undefined;
        if (ref) {
          const m = ref.match(/[:.]([a-zA-Z][\w-]*)$/);
          if (m) searchTerm = m[1];
        }
        const log = fetchGitLog({ cwd: repoRoot, file, searchTerm });
        res.setHeader("Content-Type", "application/json");
        res.end(JSON.stringify(log));
      });
    },
  };
}
```

```typescript
import type { Plugin } from "vite";
import { readFileSync, readdirSync, statSync, existsSync } from "node:fs";
import { join, relative } from "node:path";

type RelatedDoc = {
  path: string;
  lineNumber: number;
  snippet: string;
};

const SEARCH_PATHS_DEFAULT = [
  "docs/proposals/taxonomy-engine",
  "docs/proposals/taxonomy-web",
];

export function searchRelatedDocs(opts: {
  repoRoot: string;
  term: string;
  searchPaths?: string[];
  maxResults?: number;
}): RelatedDoc[] {
  const paths = opts.searchPaths ?? SEARCH_PATHS_DEFAULT;
  const max = opts.maxResults ?? 30;
  const matches: RelatedDoc[] = [];
  const term = opts.term;
  const regex = new RegExp(`\\b${escapeRegex(term)}\\b`);

  for (const dir of paths) {
    const abs = join(opts.repoRoot, dir);
    if (!existsSync(abs)) continue;
    walkMd(abs, opts.repoRoot, regex, matches, max);
    if (matches.length >= max) break;
  }
  return matches.slice(0, max);
}

function escapeRegex(s: string): string {
  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}

function walkMd(dir: string, repoRoot: string, regex: RegExp, out: RelatedDoc[], max: number): void {
  if (out.length >= max) return;
  const entries = readdirSync(dir);
  for (const name of entries) {
    if (out.length >= max) break;
    const p = join(dir, name);
    const s = statSync(p);
    if (s.isDirectory()) {
      walkMd(p, repoRoot, regex, out, max);
    } else if (s.isFile() && /\.(md|mdx)$/.test(name)) {
      const text = readFileSync(p, "utf8");
      const lines = text.split("\n");
      for (let i = 0; i < lines.length; i++) {
        if (regex.test(lines[i])) {
          out.push({
            path: relative(repoRoot, p),
            lineNumber: i + 1,
            snippet: lines[i].trim().slice(0, 200),
          });
          if (out.length >= max) return;
        }
      }
    }
  }
}

export function relatedDocsApiPlugin(repoRoot: string): Plugin {
  return {
    name: "taxonomy-web:related-docs-api",
    configureServer(server) {
      server.middlewares.use("/api/related-docs", (req, res, next) => {
        if (req.method !== "GET") {
          next();
          return;
        }
        const url = new URL(req.url ?? "/", "http://localhost");
        const term = url.searchParams.get("term");
        if (!term) {
          res.statusCode = 400;
          res.end("term query required");
          return;
        }
        const matches = searchRelatedDocs({ repoRoot, term });
        res.setHeader("Content-Type", "application/json");
        res.end(JSON.stringify(matches));
      });
    },
  };
}
```

```typescript
import type { Plugin } from "vite";
import { execSync } from "node:child_process";
import { readFileSync, existsSync } from "node:fs";
import { join } from "node:path";
import { parse, diff } from "taxonomy-engine";
import type { DiffResult } from "taxonomy-engine";

export function computeFileDiff(opts: {
  repoRoot: string;
  file: string;
  base: string;
}): DiffResult | undefined {
  const absPath = join(opts.repoRoot, opts.file);
  if (!existsSync(absPath)) return undefined;

  let baseText: string | undefined;
  try {
    baseText = execSync(`git show ${opts.base}:${opts.file}`, {
      cwd: opts.repoRoot,
      encoding: "utf8",
      stdio: ["pipe", "pipe", "pipe"],
    });
  } catch {
    return undefined;
  }
  const currentText = readFileSync(absPath, "utf8");

  const baseIR = parse(baseText);
  const currentIR = parse(currentText);
  if ("error" in baseIR || "error" in currentIR) return undefined;

  return diff(baseIR.ir, currentIR.ir);
}

export function gitDiffApiPlugin(repoRoot: string): Plugin {
  return {
    name: "taxonomy-web:git-diff-api",
    configureServer(server) {
      server.middlewares.use("/api/git-diff", (req, res, next) => {
        if (req.method !== "GET") {
          next();
          return;
        }
        const url = new URL(req.url ?? "/", "http://localhost");
        const file = url.searchParams.get("file");
        const base = url.searchParams.get("base") ?? "main";
        if (!file) {
          res.statusCode = 400;
          res.end("file query required");
          return;
        }
        const result = computeFileDiff({ repoRoot, file, base });
        res.setHeader("Content-Type", "application/json");
        res.end(JSON.stringify(result ?? null));
      });
    },
  };
}
```

```typescript
import { parseRef } from "taxonomy-engine";
import type { StaticTaxonomy } from "../types/taxonomy";

export function RelatedRulesPanel({ selectedRef, taxonomy }: { selectedRef: string; taxonomy: StaticTaxonomy }) {
  let ref;
  try { ref = parseRef(selectedRef); } catch { return null; }
  if (ref.kind !== "entry" && ref.kind !== "axis") return null;

  const matches: Array<{ relation: string; ruleIdx: number; rule: Record<string, string> }> = [];
  for (const [relName, rel] of Object.entries(taxonomy.relations)) {
    rel.rules.forEach((rule, idx) => {
      const values = Object.values(rule);
      if (ref.kind === "entry" && values.includes(ref.entry)) {
        matches.push({ relation: relName, ruleIdx: idx, rule });
      } else if (ref.kind === "axis") {
        if (rel.from.includes(ref.axis) || rel.to.includes(ref.axis)) {
          matches.push({ relation: relName, ruleIdx: idx, rule });
        }
      }
    });
  }

  if (matches.length === 0) {
    return <div className="text-xs text-gray-500">관련 relation rules 없음</div>;
  }

  return (
    <ul className="space-y-1 text-xs">
      {matches.map((m, i) => (
        <li key={i} className="font-mono">
          <span className="text-purple-700">{m.relation}#{m.ruleIdx}</span>{" "}
          <span className="text-gray-600">{JSON.stringify(m.rule)}</span>
        </li>
      ))}
    </ul>
  );
}
```

```typescript
  const matches = taxonomy.constraints.filter((c) => {

    if (ref.kind === "axis") return ruleStr.includes(ref.axis);
    if (ref.kind === "entry") return ruleStr.includes(ref.entry);
    return false;
  });

  if (matches.length === 0) {
    return <div className="text-xs text-gray-500">관련 constraints 없음</div>;
  }

  return (
    <ul className="space-y-1 text-xs">
      {matches.map((c) => (
        <li key={c.id} className="font-mono">
          <span className="text-orange-700">{c.id}</span> ({c.kind}, {c.severity})
        </li>
      ))}
    </ul>
  );
}
```

```typescript
import { parseRef } from "taxonomy-engine";
import type { StaticTaxonomy } from "../types/taxonomy";

export function SiblingEntriesView({ selectedRef, taxonomy, onSelectRef }: {
  selectedRef: string;
  taxonomy: StaticTaxonomy;
  onSelectRef: (ref: string) => void;
}) {
  let ref;
  try { ref = parseRef(selectedRef); } catch { return null; }
  if (ref.kind !== "entry") return null;

  const axis = taxonomy.axes[ref.axis];
  if (!axis) return null;
  const siblings = axis.entries.filter((e) => e.id !== ref.entry);

  if (siblings.length === 0) {
    return <div className="text-xs text-gray-500">형제 없음</div>;
  }

  return (
    <ul className="space-y-0.5 text-xs">
      {siblings.map((e) => (
        <li key={e.id}>
          <button
            className="font-mono text-blue-600 hover:underline"
            onClick={() => onSelectRef(`${taxonomy.taxonomy}:${ref.axis}.${e.id}`)}
          >
            {e.id}
          </button>
          <span className="text-gray-400 ml-2">#{e.address}</span>
        </li>
      ))}
    </ul>
  );
}
```

```typescript
import { useEffect, useState } from "react";
import type { StaticTaxonomy } from "../types/taxonomy";
import type { Feedback } from "../types/feedback";
import type { TaxonomyIndex } from "../types/taxonomy";
import { loadTaxonomy, loadTaxonomyIndex } from "../data/loadStaticTaxonomies";
import { fetchFeedbacks } from "../data/feedbacksApi";
import { TaxonomyTree } from "../components/TaxonomyTree";
import { RefBar } from "../components/RefBar";
import { FeedbackPanel } from "../components/FeedbackPanel";
import { UnitDetailPanel } from "../components/UnitDetailPanel";
import { RelatedRulesPanel } from "../components/RelatedRulesPanel";
import { RelatedConstraintsPanel } from "../components/RelatedConstraintsPanel";
import { SiblingEntriesView } from "../components/SiblingEntriesView";
import { GitLogPanel } from "../components/GitLogPanel";
import { RelatedDocsPanel } from "../components/RelatedDocsPanel";
import { DiffContextPanel } from "../components/DiffContextPanel";
import { FeedbackHistoryPanel } from "../components/FeedbackHistoryPanel";
import { Collapsible } from "../components/Collapsible";

export function TaxonomyViewPage({ file }: { file: string }) {
  const [tax, setTax] = useState<StaticTaxonomy | null>(null);
  const [yamlFile, setYamlFile] = useState<string>("");
  const [selectedRef, setSelectedRef] = useState("");
  const [feedbacks, setFeedbacks] = useState<Feedback[]>([]);

  useEffect(() => {
    loadTaxonomy(file).then(setTax);
  }, [file]);

  // taxonomy → yaml 경로 추정 (index.json의 file 필드 활용)
  useEffect(() => {
    loadTaxonomyIndex().then((idx: TaxonomyIndex) => {
      const t = idx.taxonomies.find((x) => x.file === file);
      if (t) setYamlFile(`docs/taxonomies/${t.name}.yaml`);
    });
  }, [file]);

  const refresh = () => fetchFeedbacks().then(setFeedbacks);
  useEffect(() => { refresh(); }, []);

  if (!tax) return <div>로딩 중...</div>;

  return (
    <div className="flex gap-2 h-full">
      {/* Tree */}
      <div className="flex-1 overflow-auto p-3 border-r">
        <h2 className="font-bold text-lg mb-2">{tax.taxonomy} v{tax.version}</h2>
        {tax.description && <p className="text-xs text-gray-600 mb-3">{tax.description}</p>}
        <TaxonomyTree taxonomy={tax} onSelectRef={setSelectedRef} />
      </div>

      {/* Context Panel */}
      <div className="flex-1 overflow-auto p-3 border-r space-y-2">
        <RefBar value={selectedRef} onChange={setSelectedRef} />
        {selectedRef && (
          <>
            <Collapsible title="단위 상세" defaultOpen>
              <UnitDetailPanel selectedRef={selectedRef} taxonomy={tax} />
            </Collapsible>
            <Collapsible title="관련 Relation Rules">
              <RelatedRulesPanel selectedRef={selectedRef} taxonomy={tax} />
            </Collapsible>
            <Collapsible title="관련 Constraints">
              <RelatedConstraintsPanel selectedRef={selectedRef} taxonomy={tax} />
            </Collapsible>
            <Collapsible title="형제 Entries">
              <SiblingEntriesView selectedRef={selectedRef} taxonomy={tax} onSelectRef={setSelectedRef} />
            </Collapsible>
            <Collapsible title="Git Log (이 yaml)">
              {yamlFile && <GitLogPanel file={yamlFile} selectedRef={selectedRef} />}
            </Collapsible>
            <Collapsible title="관련 문서 (plans/concepts)">
              <RelatedDocsPanel selectedRef={selectedRef} />
            </Collapsible>
            <Collapsible title="main 대비 변경">
              {yamlFile && <DiffContextPanel file={yamlFile} />}
            </Collapsible>
            <Collapsible title="이 ref의 모든 피드백" defaultOpen>
              <FeedbackHistoryPanel selectedRef={selectedRef} feedbacks={feedbacks} />
            </Collapsible>
          </>
        )}
      </div>

      {/* Feedback Form + Active */}
      <div className="w-1/3 overflow-auto p-3">
        <FeedbackPanel selectedRef={selectedRef} feedbacks={feedbacks} onChanged={refresh} />
      </div>
    </div>
  );
}
```
