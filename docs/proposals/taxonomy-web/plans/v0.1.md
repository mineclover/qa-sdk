`scripts/taxonomy-web-build-data.mjs`:

```javascript
#!/usr/bin/env node
import { readFileSync, writeFileSync, readdirSync, mkdirSync, existsSync } from "node:fs";
import { join, dirname, basename } from "node:path";
import { fileURLToPath } from "node:url";
import { parse } from "taxonomy-engine";

const __dirname = dirname(fileURLToPath(import.meta.url));
const TAX_DIR = join(__dirname, "..", "docs", "taxonomies");
const OUT_DIR = join(__dirname, "..", "packages", "taxonomy-web", "public", "data");

mkdirSync(OUT_DIR, { recursive: true });

const files = readdirSync(TAX_DIR).filter((f) => f.endsWith(".yaml"));
const index = [];

for (const file of files) {
  const text = readFileSync(join(TAX_DIR, file), "utf8");
  const r = parse(text);
  if ("error" in r) {
    console.error(`✖ parse 실패: ${file} — ${r.error.message}`);
    process.exit(1);
  }
  const ir = r.ir;
  // Map → object 직렬화
  const serialized = {
    taxonomy: ir.taxonomy,
    version: ir.version,
    description: ir.description,
    contentHash: ir.contentHash,
    axes: Object.fromEntries(ir.axes),
    relations: Object.fromEntries(ir.relations),
    constraints: ir.constraints,
    imports: ir.imports,
  };
  const outName = basename(file, ".yaml") + ".json";
  writeFileSync(join(OUT_DIR, outName), JSON.stringify(serialized, null, 2));
  index.push({
    name: ir.taxonomy,
    file: outName,
    version: ir.version,
    contentHash: ir.contentHash,
    axesCount: ir.axes.size,
    relationsCount: ir.relations.size,
    constraintsCount: ir.constraints.length,
  });
}

writeFileSync(join(OUT_DIR, "index.json"), JSON.stringify({ taxonomies: index }, null, 2));
console.log(`✔ ${files.length} taxonomies → ${OUT_DIR}`);
```

`plugins/feedback-api.ts`:

```typescript
import type { Plugin } from "vite";
import { readFileSync, writeFileSync, appendFileSync, existsSync } from "node:fs";
import type { IncomingMessage, ServerResponse } from "node:http";

type Feedback = {
  id: string;
  ref: string;
  tag: string;
  status: string;
  text: string;
  createdAt: string;
  updatedAt: string;
  history: Array<{ at: string; from: string; to: string; note?: string }>;
};

export function readFeedbacks(jsonlPath: string): Feedback[] {
  if (!existsSync(jsonlPath)) return [];
  return readFileSync(jsonlPath, "utf8")
    .split("\n")
    .filter(Boolean)
    .map((l) => JSON.parse(l) as Feedback);
}

export function appendFeedback(jsonlPath: string, feedback: Feedback): void {
  appendFileSync(jsonlPath, JSON.stringify(feedback) + "\n");
}

export function patchFeedback(
  jsonlPath: string,
  id: string,
  patch: Partial<Feedback>,
): Feedback | undefined {
  const all = readFeedbacks(jsonlPath);
  let updated: Feedback | undefined;
  const newLines = all.map((f) => {
    if (f.id !== id) return JSON.stringify(f);
    updated = { ...f, ...patch, updatedAt: new Date().toISOString() };
    return JSON.stringify(updated);
  });
  if (updated) {
    writeFileSync(jsonlPath, newLines.join("\n") + "\n");
  }
  return updated;
}

async function readBody(req: IncomingMessage): Promise<string> {
  return new Promise((resolve, reject) => {
    const chunks: Buffer[] = [];
    req.on("data", (c) => chunks.push(c));
    req.on("end", () => resolve(Buffer.concat(chunks).toString("utf8")));
    req.on("error", reject);
  });
}

export function feedbackApiPlugin(jsonlPath: string): Plugin {
  return {
    name: "taxonomy-web:feedback-api",
    configureServer(server) {
      server.middlewares.use("/api/feedbacks", async (req: IncomingMessage, res: ServerResponse, next) => {
        try {
          if (req.method === "GET") {
            const feedbacks = readFeedbacks(jsonlPath);
            res.setHeader("Content-Type", "application/json");
            res.end(JSON.stringify(feedbacks));
            return;
          }
          if (req.method === "POST") {
            const body = await readBody(req);
            const feedback = JSON.parse(body) as Feedback;
            appendFeedback(jsonlPath, feedback);
            res.statusCode = 201;
            res.setHeader("Content-Type", "application/json");
            res.end(JSON.stringify(feedback));
            return;
          }
          if (req.method === "PATCH") {
            const body = await readBody(req);
            const { id, ...patch } = JSON.parse(body);
            const updated = patchFeedback(jsonlPath, id, patch);
            if (!updated) {
              res.statusCode = 404;
              res.end();
              return;
            }
            res.statusCode = 200;
            res.setHeader("Content-Type", "application/json");
            res.end(JSON.stringify(updated));
            return;
          }
          next();
        } catch (e) {
          res.statusCode = 500;
          res.end(String((e as Error).message));
        }
      });
    },
  };
}
```

```typescript
let counter = 0;
let lastDate = "";

export function newFeedbackId(now: Date = new Date()): string {
  const date = now.toISOString().slice(0, 10).replace(/-/g, "");
  if (date !== lastDate) {
    counter = 0;
    lastDate = date;
  }
  counter++;
  return `f-${date}-${String(counter).padStart(3, "0")}`;
}

// 테스트용 reset
export function _resetCounter(): void {
  counter = 0;
  lastDate = "";
}
```

```typescript
import type { TaxonomyRef } from "taxonomy-engine";

export function isValidRef(input: string): boolean {
  try {
    parseRef(input);
    return true;
  } catch {
    return false;
  }
}

export function displayRef(input: string): string {
  try {
    const ref = parseRef(input);
    if (ref.kind === "entry") return `${ref.taxonomy}:${ref.axis}.${ref.entry}`;
    if (ref.kind === "axis") return `${ref.taxonomy}:${ref.axis}`;
    if (ref.kind === "taxonomy") return ref.taxonomy;
    return input;
  } catch {
    return input;
  }
}

export function refToCoordTuple(input: string): { taxonomy: string; rest: string } | undefined {
  try {
    const ref = parseRef(input);
    return { taxonomy: ref.taxonomy, rest: JSON.stringify(ref) };
  } catch {
    return undefined;
  }
}
```

```typescript
import type { Feedback, FeedbackTag, FeedbackStatus } from "../types/feedback";

export type FeedbackFilter = {
  ref?: string;
  tags?: FeedbackTag[];
  statuses?: FeedbackStatus[];
  taxonomy?: string;   // ref가 이 taxonomy로 시작
};

export function filterFeedback(items: Feedback[], filter: FeedbackFilter): Feedback[] {
  return items.filter((f) => {
    if (filter.ref && f.ref !== filter.ref) return false;
    if (filter.tags && filter.tags.length > 0 && !filter.tags.includes(f.tag)) return false;
    if (filter.statuses && filter.statuses.length > 0 && !filter.statuses.includes(f.status)) return false;
    if (filter.taxonomy && !f.ref.startsWith(filter.taxonomy)) return false;
    return true;
  });
}
```

```typescript
export function patchFeedback(
  jsonlPath: string,
  id: string,
  patch: Partial<Feedback>,
): Feedback | undefined {
  const all = readFeedbacks(jsonlPath);
  let updated: Feedback | undefined;
  const newLines = all.map((f) => {
    if (f.id !== id) return JSON.stringify(f);
    // history는 append 모드
    const newHistory = patch.history ? [...f.history, ...patch.history] : f.history;
    updated = { ...f, ...patch, history: newHistory, updatedAt: new Date().toISOString() };
    return JSON.stringify(updated);
  });
  if (updated) {
    writeFileSync(jsonlPath, newLines.join("\n") + "\n");
  }
  return updated;
}
```

```typescript
import { useEffect, useState } from "react";
import type { Feedback, FeedbackTag, FeedbackStatus } from "../types/feedback";
import { FEEDBACK_TAGS, FEEDBACK_STATUSES } from "../types/feedback";
import { fetchFeedbacks } from "../data/feedbacksApi";
import { filterFeedback } from "../lib/filterFeedback";
import { FeedbackList } from "../components/FeedbackList";

export function FeedbacksPage() {
  const [items, setItems] = useState<Feedback[]>([]);
  const [tags, setTags] = useState<FeedbackTag[]>([]);
  const [statuses, setStatuses] = useState<FeedbackStatus[]>([]);

  useEffect(() => { fetchFeedbacks().then(setItems); }, []);

  const filtered = filterFeedback(items, { tags: tags.length ? tags : undefined, statuses: statuses.length ? statuses : undefined });

  return (
    <div className="p-4">
      <h2 className="font-bold text-lg mb-3">전체 피드백 ({filtered.length}/{items.length})</h2>
      <div className="flex gap-4 mb-4 text-sm">
        <div>
          <strong>tag:</strong>
          {FEEDBACK_TAGS.map((t) => (
            <label key={t} className="ml-2">
              <input
                type="checkbox"
                checked={tags.includes(t)}
                onChange={(e) =>
                  setTags((prev) => (e.target.checked ? [...prev, t] : prev.filter((x) => x !== t)))
                }
              />{" "}{t}
            </label>
          ))}
        </div>
        <div>
          <strong>status:</strong>
          {FEEDBACK_STATUSES.map((s) => (
            <label key={s} className="ml-2">
              <input
                type="checkbox"
                checked={statuses.includes(s)}
                onChange={(e) =>
                  setStatuses((prev) => (e.target.checked ? [...prev, s] : prev.filter((x) => x !== s)))
                }
              />{" "}{s}
            </label>
          ))}
        </div>
      </div>
      <FeedbackList items={filtered} />
    </div>
  );
}
```
