import { describe, it, expect } from "vitest";
import { parse } from "../../src/parser/index.js";
import { diff } from "../../src/diff/index.js";

function ir(y: string) {
  const r = parse(y);
  if (!("ir" in r)) throw new Error();
  return r.ir;
}

describe("diff breakingLevel", () => {
  it("entry-removed → major", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }, { id: y, address: 1, lifecycle: approved }] } }`);
    const b = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    expect(diff(a, b).breakingLevel).toBe("major");
  });

  it("entry-added → minor", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    const b = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }, { id: y, address: 1, lifecycle: approved }] } }`);
    expect(diff(a, b).breakingLevel).toBe("minor");
  });

  it("entry-lifecycle-changed → patch", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: tbd, placeholderReason: "y" }] } }`);
    const b = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    expect(diff(a, b).breakingLevel).toBe("patch");
  });

  it("capacity 감소 → major, 증가 → minor", () => {
    const base = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, capacity: 16, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    const smaller = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, capacity: 8, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    const bigger = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, capacity: 32, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    expect(diff(base, smaller).breakingLevel).toBe("major");
    expect(diff(base, bigger).breakingLevel).toBe("minor");
  });

  it("major + minor + patch 동시 → major (최고 우선)", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }, { id: y, address: 1, lifecycle: approved }] } }`);
    const b = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: tbd, placeholderReason: "y" }, { id: z, address: 2, lifecycle: approved }] } }`);
    // y removed (major), z added (minor), x lifecycle changed (patch)
    expect(diff(a, b).breakingLevel).toBe("major");
  });

  it("same → none", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { a: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    expect(diff(a, a).breakingLevel).toBe("none");
  });

  it("relation-rule-removed → major", () => {
    const a = ir(`taxonomy: t
version: 1
axes:
  a:
    lifecycle: approved
    entries: [{ id: x, address: 0, lifecycle: approved }, { id: y, address: 1, lifecycle: approved }]
relations:
  r:
    from: a
    to: a
    kind: depends-on
    rules:
      - { a: x }
      - { a: y }`);
    const b = ir(`taxonomy: t
version: 1
axes:
  a:
    lifecycle: approved
    entries: [{ id: x, address: 0, lifecycle: approved }, { id: y, address: 1, lifecycle: approved }]
relations:
  r:
    from: a
    to: a
    kind: depends-on
    rules:
      - { a: x }`);
    expect(diff(a, b).breakingLevel).toBe("major");
  });
});
```

import { describe, it, expect } from "vitest";
import { parse } from "../../src/parser/index.js";
import { migrationPlan } from "../../src/migration/index.js";

function ir(y: string) {
  const r = parse(y);
  if (!("ir" in r)) throw new Error();
  return r.ir;
}

describe("migrationPlan", () => {
  it("address 같고 id 다르면 rename 후보", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { layer: { lifecycle: approved, entries: [{ id: cue, address: 3, lifecycle: approved }] } }`);
    const b = ir(`taxonomy: t
version: 1
axes: { layer: { lifecycle: approved, entries: [{ id: moment, address: 3, lifecycle: approved }] } }`);
    const plan = migrationPlan(a, b);
    expect(plan.idRenames).toEqual([
      { axis: "layer", fromId: "cue", toId: "moment", address: 3 },
    ]);
    expect(plan.addressConflicts).toEqual([]);
  });

  it("address도 id도 다르면 rename 아님 (add/remove로 diff가 처리)", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { l: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    const b = ir(`taxonomy: t
version: 1
axes: { l: { lifecycle: approved, entries: [{ id: y, address: 1, lifecycle: approved }] } }`);
    const plan = migrationPlan(a, b);
    expect(plan.idRenames).toEqual([]);
  });

  it("같은 변경 → 빈 plan", () => {
    const a = ir(`taxonomy: t
version: 1
axes: { l: { lifecycle: approved, entries: [{ id: x, address: 0, lifecycle: approved }] } }`);
    const plan = migrationPlan(a, a);
    expect(plan.idRenames).toEqual([]);
    expect(plan.addressConflicts).toEqual([]);
    expect(plan.manualReview).toEqual([]);
  });

  it("capacity 감소 + entries 초과 → manualReview 보고", () => {
    const a = ir(`taxonomy: t
version: 1
axes:
  l:
    lifecycle: approved
    capacity: 16
    entries:
      - { id: a, address: 0, lifecycle: approved }
      - { id: b, address: 1, lifecycle: approved }
      - { id: c, address: 2, lifecycle: approved }`);
    const b = ir(`taxonomy: t
version: 1
axes:
  l:
    lifecycle: approved
    capacity: 2
    entries:
      - { id: a, address: 0, lifecycle: approved }
      - { id: b, address: 1, lifecycle: approved }
      - { id: c, address: 2, lifecycle: approved }`);
    const plan = migrationPlan(a, b);
    expect(plan.manualReview.some((m) => m.includes("capacity"))).toBe(true);
  });
});
```
